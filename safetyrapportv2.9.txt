  let currentStep = 1;
  const totalSteps = 9;

  const visitedSteps = new Set([1]);

  function setDeviceMode(mode, event) {
    const btnComp = document.getElementById('btnComp');
    const btnPhone = document.getElementById('btnPhone');

    btnComp.classList.remove('active');
    btnPhone.classList.remove('active');

    if (mode === 'phone') {
      document.body.classList.add('device-phone');
      document.body.classList.remove('device-computer');
      if(event && event.target) event.target.classList.add('active');
      else btnPhone.classList.add('active');
    } else {
      document.body.classList.add('device-computer');
      document.body.classList.remove('device-phone');
      if(event && event.target) event.target.classList.add('active');
      else btnComp.classList.add('active');
    }
  }

  function goToStep(step) {
    if (step < 1 || step > totalSteps) return;
    
    visitedSteps.add(step);

    document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
    document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));

    currentStep = step;
    document.getElementById(`step-${step}`).classList.add('active');
    document.getElementById(`tab-btn-${step}`).classList.add('active');

    document.getElementById('prevBtn').style.display = currentStep === 1 ? 'none' : 'inline-block';
    if (currentStep === totalSteps) {
      document.getElementById('nextBtn').classList.add('hidden');
      document.getElementById('submitBtn').classList.remove('hidden');
      checkDocumentationRequirements();
    } else {
      document.getElementById('nextBtn').classList.remove('hidden');
      document.getElementById('submitBtn').classList.add('hidden');
    }

    updateTabStatuses();
  }

  function changeStep(delta) {
    goToStep(currentStep + delta);
  }

  function getTabStatus(stepNum) {
    const stepEl = document.getElementById(`step-${stepNum}`);
    if (!stepEl) return 'unread';

    const reqInputs = Array.from(stepEl.querySelectorAll('input.field-req, textarea.field-req, select.field-req, input[required], textarea[required]'))
      .filter(el => !el.closest('.hidden'));

    const optInputs = Array.from(stepEl.querySelectorAll('input.field-opt, textarea.field-opt, select.field-opt, input[type="text"]:not(.field-req), textarea:not(.field-req)'))
      .filter(el => !el.closest('.hidden') && el.id !== 'selectedRiskScore');

    let hasEmptyRequired = false;
    reqInputs.forEach(input => {
      if (!input.value || input.value.trim() === '') {
        hasEmptyRequired = true;
      }
    });

    if (stepNum === 6) {
      const hasRisk = document.querySelector('input[name="hasRiskAssessment"]:checked').value === 'yes';
      const score = document.getElementById('selectedRiskScore').value;
      if (hasRisk && !score) {
        hasEmptyRequired = true;
      }
    }

    if (hasEmptyRequired) {
      return visitedSteps.has(stepNum) ? 'red' : 'unread';
    }

    let hasEmptyOptional = false;
    optInputs.forEach(input => {
      if (input.type === 'file') {
        if (!input.files || input.files.length === 0) hasEmptyOptional = true;
      } else if (!input.value || input.value.trim() === '') {
        hasEmptyOptional = true;
      }
    });

    if (!visitedSteps.has(stepNum)) {
      return hasEmptyOptional ? 'unread' : 'green';
    }

    if (hasEmptyOptional) {
      return 'yellow';
    }

    return 'green';
  }

  function updateTabStatuses() {
    for (let i = 1; i <= totalSteps; i++) {
      const status = getTabStatus(i);
      const tabBtn = document.getElementById(`tab-btn-${i}`);
      
      tabBtn.classList.remove('status-unread', 'status-red', 'status-yellow', 'status-green');
      tabBtn.classList.add(`status-${status}`);
    }
  }

  let mediaCount = 0;
  function handleDocChoice(choice) {
    const uploadSec = document.getElementById('docUploadSection');
    if (choice === 'no') {
      uploadSec.classList.add('hidden');
    } else {
      uploadSec.classList.remove('hidden');
      if (mediaCount === 0) addMediaCard();
    }
    checkDocumentationRequirements();
    updateTabStatuses();
  }

  function addMediaCard() {
    mediaCount++;
    const container = document.getElementById('mediaContainer');
    const item = document.createElement('div');
    item.className = 'dynamic-item-card';
    item.id = 'media_' + mediaCount;
    item.innerHTML = `
      <button type="button" class="remove-card-btn" onclick="removeElement('media_${mediaCount}')">&times;</button>
      <div class="form-group">
        <label>Vælg billede ${mediaCount}</label>
        <input type="file" class="media-file-input field-opt" accept="image/*">
      </div>
      <div class="form-group" style="margin-bottom:0;">
        <label>Kommentar til billede ${mediaCount}</label>
        <input type="text" class="media-comment field-opt" placeholder="f.eks. Bule på højre vinge, ridse på GSE osv.">
      </div>
    `;
    container.appendChild(item);
    updateTabStatuses();
  }

  function checkDocumentationRequirements() {
    const docRadio = document.querySelector('input[name="hasDocumentation"]:checked');
    const chosenNoDoc = docRadio && docRadio.value === 'no';

    const acDamageVal = document.getElementById('damageAircraft').value;
    const hasAcDamage = (acDamageVal === 'Yes - Minor' || acDamageVal === 'Yes - Major');
    
    const gseRadio = document.querySelector('input[name="hasGSE"]:checked');
    const hasGSE = gseRadio && gseRadio.value === 'yes';

    const injuriesVal = document.getElementById('injuries').value;
    const hasInjuries = (injuriesVal !== 'No');

    const requiresReason = chosenNoDoc || hasAcDamage || hasGSE || hasInjuries;

    const reasonGroup = document.getElementById('noDocReasonGroup');
    if (requiresReason && chosenNoDoc) {
      reasonGroup.classList.remove('hidden');
    } else {
      reasonGroup.classList.add('hidden');
    }
  }

  function removeElement(id) {
    const elem = document.getElementById(id);
    if (elem) elem.remove();
    updateTabStatuses();
  }

  function selectCard(cardElement, groupName) {
    const parent = cardElement.closest('.report-type-grid');
    parent.querySelectorAll('.report-type-card').forEach(card => card.classList.remove('selected'));
    cardElement.classList.add('selected');
    const radio = cardElement.querySelector('input[type="radio"]');
    if (radio) radio.checked = true;
    updateTabStatuses();
  }

  let recognition;
  let isRecording = false;

  if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRecognition();
    recognition.lang = 'da-DK';
    recognition.continuous = true;
    recognition.interimResults = true;

    recognition.onresult = (event) => {
      let transcript = '';
      for (let i = event.resultIndex; i < event.results.length; i++) {
        transcript += event.results[i][0].transcript;
      }
      const reportTextElem = document.getElementById('reportText');
      reportTextElem.value += (reportTextElem.value ? ' ' : '') + transcript;
      updateTabStatuses();
    };
  }

  function toggleSpeechRecognition() {
    if (!recognition) return;
    const btn = document.getElementById('startVoiceBtn');
    const status = document.getElementById('voiceStatus');

    if (!isRecording) {
      recognition.start();
      isRecording = true;
      btn.innerHTML = '🛑 Stop indtalelse';
      status.innerText = 'Lytter... Tal nu.';
    } else {
      recognition.stop();
      isRecording = false;
      btn.innerHTML = '🎤 Indtal besked (Tal-til-tekst)';
      status.innerText = 'Indtalelse stoppet.';
    }
  }

  document.getElementById('safetyReportForm').addEventListener('submit', function(e) {
    e.preventDefault();

    for (let i = 1; i <= totalSteps; i++) {
      visitedSteps.add(i);
    }
    updateTabStatuses();

    let redSteps = [];
    for (let i = 1; i <= totalSteps; i++) {
      if (getTabStatus(i) === 'red') {
        redSteps.push(i);
      }
    }

    if (redSteps.length > 0) {

      goToStep(redSteps[0]);
      return;
    }

    const title = document.getElementById('reportTitle').value.trim();
    const time = document.getElementById('occurrenceTime').value;
    const location = document.getElementById('occurrenceLocation').value.trim();
    const reportText = document.getElementById('reportText').value.trim();

    const isAnon = document.querySelector('input[name="isAnonymous"]:checked').value === 'yes';
    const reportType = document.querySelector('input[name="reportType"]:checked').value;

    const supName = document.getElementById('supervisorName').value;

    const repName = document.getElementById('reporterName').value;

    const hasAircraft = document.querySelector('input[name="hasAircraft"]:checked').value === 'yes';
    const airlineOwner = document.getElementById('airlineOwner').value;
    const flightNo = document.getElementById('flightNumber').value;
    const acReg = document.getElementById('aircraftRegistration').value;
    const damage = document.getElementById('damageAircraft').value;
    const damageDetails = document.getElementById('aircraftDamageDetails').value;
    const delay = document.getElementById('flightDelay').value;
    const delayH = document.getElementById('delayHours').value;
    const delayM = document.getElementById('delayMinutes').value;

    const injuries = document.getElementById('injuries').value;
    const riskHurtRadio = document.querySelector('input[name="riskOfHurt"]:checked');
    const riskHurt = riskHurtRadio ? riskHurtRadio.value : 'no';

    let gseDetails = [];
    const gseRadio = document.querySelector('input[name="hasGSE"]:checked');
    if (gseRadio && gseRadio.value === 'yes') {
      document.querySelectorAll('#gseContainer .dynamic-item-card').forEach(item => {
        const type = item.querySelector('.gse-type').value;
        const id = item.querySelector('.gse-id').value;
        if (type || id) gseDetails.push(`${type} (ID: ${id})`);
      });
    }

    const riskScore = document.getElementById('selectedRiskScore').value || "Ingen";

    let imageComments = [];
    document.querySelectorAll('#mediaContainer .dynamic-item-card').forEach((item, index) => {
      const fileInput = item.querySelector('.media-file-input');
      const comment = item.querySelector('.media-comment').value;
      const fileName = fileInput.files[0] ? fileInput.files[0].name : "Billede " + (index + 1);
      imageComments.push(`${fileName}: ${comment || "Ingen kommentar"}`);
    });

    const noDocReason = document.getElementById('noDocReason').value;

    let emailBody = `--- SAFETY INCIDENT REPORT ---\n\n`;
    emailBody += `1. RAPPORTERINGSTYPE\nKategori: ${reportType}\n\n`;

    emailBody += `2. ANONYMITET & KONTAKT\nAnonym: ${isAnon ? 'Ja' : 'Nej'}\n`;
    if (isAnon) {
      emailBody += `Supervisor: ${supName} (${supContact})\n\n`;
    } else {
      emailBody += `Rapportør: ${repName} (${repContact})\nAfdeling: ${dept}\n\n`;
    }

    emailBody += `3. STAMDATA & PLACERING\nOverskrift: ${title}\nTidspunkt (24t): ${time}\nPlacering (Apron Syd): ${location}\n\n`;

    emailBody += `4. AIRLINE & FLIGHT\nFly involveret: ${hasAircraft ? 'Ja' : 'Nej'}\n`;
    if (hasAircraft) {
      emailBody += `Ejer / Airline: ${airlineOwner}\nFlight nr.: ${flightNo}\nRegistrering: ${acReg}\nSkade på fly: ${damage}\n`;
      if (damage === 'Yes - Minor' || damage === 'Yes - Major') {
        emailBody += `Skadesomfang: ${damageDetails}\n`;
      }
      emailBody += `Forsinkelse: ${delay}\n`;
      if (delay === 'Yes') {
        emailBody += `Varighed af delay: ${delayH} timer og ${delayM} minutter\n`;
      }
      emailBody += `\n`;
    }

    emailBody += `5. GROUND OPERATIONS\nPersonskade: ${injuries}\nRisiko for personskade: ${riskHurt}\nGSE involveret: ${gseDetails.length > 0 ? gseDetails.join(', ') : 'Nej'}\n\n`;

    emailBody += `6. INITIAL RISK\nRisikoscore: ${riskScore}\n\n`;

    emailBody += `8. BESKRIVELSE AF HÆNDELSEN\n${reportText}\n\n`;

    emailBody += `9. DOKUMENTATION & BILLEDER\n`;
    if (imageComments.length > 0) {
      emailBody += `Billednotater:\n` + imageComments.join('\n') + `\n`;
    }
    if (noDocReason) {
      emailBody += `Begrundelse for manglende dok.: ${noDocReason}\n`;
    }

    const recipient = "tsc@bll.dk";
    const subject = encodeURIComponent("New Safety Report: " + title);
    const body = encodeURIComponent(emailBody);

    window.location.href = `mailto:${recipient}?subject=${subject}&body=${body}`;

    setTimeout(() => {
      alert("Dine oplysninger er valideret. Mailprogrammet åbnes nu til tsc@bll.dk.");
    }, 400);
  });
